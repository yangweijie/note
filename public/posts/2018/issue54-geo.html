geo扩展库安装

2018-04-18

安装geos yum install geos
git clone https://git.osgeo.org/gitea/geos/php-geos.git
cd php-geos
./autogen.sh
./configure --with-php-config=/Applications/MAMP/bin/php/php7.1.12/bin/php-config
make # generates modules/geos.so
make install
改php.ini
